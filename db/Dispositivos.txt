-- CREAR TABLA DE DISPOSITIVOS

USE [CoreSightJRPro];
GO

CREATE TABLE dbo.dispositivos (
    id INT IDENTITY(1,1) NOT NULL,
    ip VARCHAR(15) NOT NULL,
    economico VARCHAR(6) NOT NULL,
    nombre NVARCHAR(75) NULL,
    descripcion NVARCHAR(100) NULL,
    general NVARCHAR(MAX) NULL,

    CONSTRAINT PK_dispositivos
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT UQ_dispositivos_ip
        UNIQUE (ip),

    CONSTRAINT FK_dispositivos_sucursales_economico
        FOREIGN KEY (economico)
        REFERENCES dbo.sucursales (economico)
);
GO

-- REINICIAR EL ID
DBCC CHECKIDENT ('dispositivos', RESEED, 0);
