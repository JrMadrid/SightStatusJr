-- CREAR TABLA DE INFORMES

USE [CoreSightJr];
GO

CREATE TABLE dbo.informes (
    id INT IDENTITY(1,1) NOT NULL,
    economico VARCHAR(6) NOT NULL,
    fecharealizada DATE NOT NULL,
    nombre NVARCHAR(100) NULL,
    descripcion NVARCHAR(100) NULL,
    informe VARBINARY(MAX) NOT NULL,

    CONSTRAINT PK_informes
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT FK_informes_sucursales_economico
        FOREIGN KEY (economico)
        REFERENCES dbo.sucursales (economico)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);
GO


-- REINICIAR EL ID
DBCC CHECKIDENT ('informes', RESEED, 0);
