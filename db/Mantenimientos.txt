-- CREAR TABLA DE MANTENIMIENTOS 

USE [SightStatusJr];
GO

CREATE TABLE dbo.mantenimientos (
    id INT IDENTITY(1,1) NOT NULL,
    fechaestimada DATE NOT NULL 
        CONSTRAINT DF_mantenimientos_fechaestimada DEFAULT ('2000-01-01'),
    fecharealizada DATE NULL,
    economico VARCHAR(6) NOT NULL 
        CONSTRAINT DF_mantenimientos_economico DEFAULT ('000000'),
    constancia VARBINARY(MAX) NULL,
    descripcion NVARCHAR(MAX) NULL,

    CONSTRAINT PK_mantenimientos
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT FK_mantenimientos_sucursales_economico
        FOREIGN KEY (economico)
        REFERENCES dbo.sucursales (economico)
);
GO


-- DATO ESPECIAL
-- Documentos de 2024 o mas antiguos son considerados historicos y no generan nuevos mantenimientos estimados.
-- (Opcional) Se podría dejar el primer elemento como la plantilla a utilizar para los mantenimientos
