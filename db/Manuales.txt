-- CREAR TABLA DE MANUALES

USE [CoreSightJr];
GO

CREATE TABLE dbo.manuales (
    id INT IDENTITY(1,1) NOT NULL,
    nombre NVARCHAR(100) NOT NULL 
        CONSTRAINT DF_manuales_nombre DEFAULT (''),
    manual VARBINARY(MAX) NOT NULL,
    descripcion NVARCHAR(100) NULL,

    CONSTRAINT PK_manuales
        PRIMARY KEY CLUSTERED (id)
);
GO


-- REINICIAR EL ID
DBCC CHECKIDENT ('manuales', RESEED, 0);
