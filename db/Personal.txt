-- CREAR TABLA DE PERSONAL

USE [CoreSightJrPro];
GO

CREATE TABLE dbo.personal (
    id INT IDENTITY(1,1) NOT NULL,
    nickname NVARCHAR(50) NOT NULL,
    cedula CHAR(15) NOT NULL 
        CONSTRAINT DF_personal_cedula DEFAULT ('0000-0000-00000'),
    localidad NVARCHAR(100) NULL,
    foto VARBINARY(MAX) NULL,
    fecha_nacimiento DATE NULL,
    sexo CHAR(1) NULL,
    fecha_contratacion DATE NOT NULL 
        CONSTRAINT DF_personal_fecha_contratacion DEFAULT (GETDATE()),
    descripcion NVARCHAR(MAX) NULL,
    grado_academico NVARCHAR(100) NULL,
    puesto NVARCHAR(100) NOT NULL 
        CONSTRAINT DF_personal_puesto DEFAULT ('SIN PUESTO'),
    nombre NVARCHAR(200) NULL,
    telefono VARCHAR(20) NULL,

    CONSTRAINT PK_personal
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT UQ_personal_nickname
        UNIQUE (nickname),

    CONSTRAINT FK_personal_users_nickname
        FOREIGN KEY (nickname)
        REFERENCES dbo.users (nickname)
        ON DELETE CASCADE
);
GO
