-- CREAR TABLA DE SUCURSALES

USE [SightStatusJr];
GO

CREATE TABLE dbo.sucursales (
    id INT IDENTITY(1,1) NOT NULL,
    canal NVARCHAR(50) NULL,
    nombre NVARCHAR(50) NULL,
    economico VARCHAR(6) NOT NULL,
    ingresponsable NVARCHAR(50) NOT NULL,

    CONSTRAINT PK_sucursales
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT UQ_sucursales_economico
        UNIQUE (economico),

    CONSTRAINT FK_sucursales_users_ingresponsable
        FOREIGN KEY (ingresponsable)
        REFERENCES dbo.users (nickname)
);
GO
