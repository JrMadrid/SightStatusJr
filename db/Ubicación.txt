-- CREAR TABLA DE UBICACIÓN DE LAS SUCURSALES

USE [CoreSightJrPro];
GO

CREATE TABLE dbo.ubicacion (
    id INT IDENTITY(1,1) NOT NULL,
    economico VARCHAR(6) NOT NULL,
    latitud DECIMAL(10,8) NULL,
    longitud DECIMAL(11,8) NULL,
    direccion NVARCHAR(255) NULL,
    actualizado DATETIME NOT NULL 
        CONSTRAINT DF_ubicacion_actualizado DEFAULT (GETDATE()),
    imagen VARBINARY(MAX) NULL,
    descripcion NVARCHAR(MAX) NULL,

    CONSTRAINT PK_ubicacion
        PRIMARY KEY CLUSTERED (id),

    CONSTRAINT FK_ubicacion_sucursales_economico
        FOREIGN KEY (economico)
        REFERENCES dbo.sucursales (economico)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);
GO
